def _test_shape(m, n, scale=1.0):
    torch.manual_seed(0)
    x = torch.randn(m, n) * scale
    assert torch.allclose(softmax(x), torch.softmax(x, axis=1), atol=1e-5),
        f"mismatch {m}x{n} scale={scale}"
