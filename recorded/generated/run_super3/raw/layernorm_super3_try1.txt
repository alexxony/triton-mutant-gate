def check_const(M, N, val=1.0, eps=1e-5):
    """Test a constant input (zero variance). Expected output equals the bias."""
    torch.manual_seed(0)
    x = torch.full((M, N), val, dtype=torch.float32)
    w = torch.rand(N)
    b = torch.rand(N)
    ref = torch.nn.functional.layer_norm(x, (N,), w, b, eps)
    y = layer_norm(x, w, b, eps)
    # For constant input, normalized values are zero => y = b
    assert torch.allclose(y, b, atol=1e-5, rtol=0), f"constant input mismatch {M}x{N}"
    # Also verify against PyTorch's reference implementation
    assert torch.allclose(y, ref, atol=1e-5, rtol=0), f"constant vs ref mismatch {M}x{N}"
