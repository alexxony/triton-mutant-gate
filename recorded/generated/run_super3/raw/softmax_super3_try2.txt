def _assert_close(x, msg=""):
    y = softmax(x)
    y_ref = torch.softmax(x, dim=1)
    # torch.allclose fails on NaNs; replace NaNs with a sentinel that compares equal
    y_nan = torch.isnan(y)
    y_ref_nan = torch.isnan(y_ref)
    assert torch.all(y_nan == y_ref_nan), msg + f" NaN mismatch {tuple(x.shape)}"
    # compare only the finite parts
    finite = ~y_nan
    if finite.any():
        assert torch.allclose(y[finite], y_ref[finite], atol=1e-5, rtol=1e-5), \
            msg + f" value mismatch {tuple(x.shape)}"
