def test_zero_variance():
    """When variance is zero (constant row) removing eps leads to inf/nan."""
    M, N = 2, 256          # small enough to stay well under 2M elements
    eps = 1e-5
    torch.manual_seed(123)
    const_val = -1.2345    # any constant works
    x = torch.full((M, N), const_val, dtype=torch.float32)
    w = torch.randn(N, dtype=torch.float32)
    b = torch.randn(N, dtype=torch.float32)
    ref = torch.nn.functional.layer_norm(x, (N,), w, b, eps)
    y = layer_norm(x, w, b, eps)
    # For constant input, LayerNorm output should be exactly the bias (up to tiny numerical error)
    assert torch.allclose(y, ref, atol=1e-5, rtol=0), \
        f"zero‑variance test failed: max diff = {(y - ref).abs().max()}"

# Run the new test
test_zero_variance()
