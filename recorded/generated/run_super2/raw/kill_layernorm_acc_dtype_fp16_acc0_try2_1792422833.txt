def test_cancellation_error():
    """Create a case where the true mean is zero but low‑precision (float16) accumulation
    produces a noticeable bias, causing the layer‑norm output to differ."""
    M, N = 2, 8192               # total elements = 16 384 (< 2 M)
    eps = 1e-5
    half = N // 2
    val = 7.5                     # chosen so that N*val < float16 max (≈65504) but >2048
    x = torch.full((M, N), val, dtype=torch.float32)
    x[:, half:] = -val            # first half +val, second half –val → true mean = 0
    w = torch.ones(N, dtype=torch.float32)   # weight = 1
    b = torch.zeros(N, dtype=torch.float32)  # bias = 0
    ref = torch.nn.functional.layer_norm(x, (N,), w, b, eps)
    y   = layer_norm(x, w, b, eps)
    # The mutant’s float16 accumulator yields a biased mean → output mismatch
    assert torch.allclose(y, ref, atol=1e-2, rtol=0), \
        f"float16 mean accumulation bias detected for {M}x{N}"

test_cancellation_error()
