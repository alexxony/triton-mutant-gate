def test_row_padded():

    # Allocate a wider matrix and take a left‑aligned slice → stride[0] > N*element_size
    base = torch.empty(M, N * 2, dtype=torch.float32)
    x = base[:, :N]                     # shape (M, N), stride[0] = 2*N*sizeof(float)
    w = torch.randn(N, dtype=torch.float32)
    b = torch.randn(N, dtype=torch.float32)
    ref = torch.nn.functional.layer_norm(x, (N,), w, b, eps=1e-5)
