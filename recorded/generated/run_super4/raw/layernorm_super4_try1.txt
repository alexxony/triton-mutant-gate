def _test_non_multiple_block():

def _test_eps_zero_const_row():
    """When variance is zero and eps=0 the kernel should produce NaN (0*inf)."""
    torch.manual_seed(0)
    M, N = 2, 1024
    x = torch.full((M, N), 3.0)          # constant row → var = 0
    w = torch.rand(N, dtype=torch.float32)
    b = torch.rand(N, dtype=torch.float32)
    # PyTorch reference (will be NaN)
    ref = torch.nn.functional.layer_norm(x, (N,), w, b, eps=0.0)
    y   = layer_norm(x, w, b, eps=0.0)
    # Both should be all‑NaN
    assert torch.isnan(ref).all() and torch.isnan(y).all(), \
        "eps=0 with constant row did not produce NaN"


def _test_fp16_input():
    """Float16 input with float32 weights/biases – checks casting and accumulation."""
    torch.manual_seed(0)
    M, N = 2, 1025                     # non‑multiple block size to also test tail
    x = (-2.3 + 0.5 * torch.randn(M, N, dtype=torch.float16))
    w = torch.rand(N, dtype=torch.float32)
    b = torch.rand(N, dtype=torch.float32)
    # Reference in higher precision then cast back
    ref = torch.nn.functional.layer_norm(x.float(), (N,), w, b, eps=1e-5).half()
    y   = layer_norm(x, w, b, eps=1e-5)
    assert torch.allclose(y, ref, atol=1e-3, rtol=0), \
        "float16 input test failed"
