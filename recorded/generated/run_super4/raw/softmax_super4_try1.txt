def _extra_check(m, n, scale=1.0, dtype=torch.float32, contiguous=True):
    torch.manual_seed(0)
    x = torch.randn(m, n, dtype=dtype) * scale
    if not contiguous:
        # create a non‑contiguous view by striding a larger base tensor
        base = torch.randn(m * 2, n * 2, dtype=dtype) * scale
        x = base[::2, ::2][:m, :n]
    y = softmax(x)
    y_ref = torch.softmax(x, dim=1)
    assert torch.allclose(y, y_ref, atol=1e-5, rtol=1e-5), \
        f"m={m}, n={n}, dtype={dtype}, contiguous={contiguous}, scale={scale}"
