def test_accumulator_type():
    # Use a large K to amplify the error when accumulating in float16
    M, N, K = 64, 64, 2048
    torch.manual_seed(0)
    a = torch.ones((M, K), dtype=torch.float16)   # all‑ones matrix
    b = torch.ones((K, N), dtype=torch.float16)   # all‑ones matrix
    ref = torch.matmul(a, b)                      # PyTorch accumulates in higher precision
    out = matmul(a, b)                            # Triton kernel under test
    # The mutant accumulates in float16 → large deviation from the reference
    assert torch.allclose(out, ref, atol=0.125, rtol=0), \
        f"accumulator type mismatch: max diff {torch.max(torch.abs(out - ref)).item()}"

test_accumulator_type()
