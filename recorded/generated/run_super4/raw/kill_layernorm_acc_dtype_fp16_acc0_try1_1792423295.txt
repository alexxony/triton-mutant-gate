def _test_fp16_mean_error():
    """Detect loss of precision when the mean accumulator is held in float16."""
    M, N = 1, 100_000          # well under 2 M elements
    eps = 1e-5
    shift = 1.0e3              # large values to amplify rounding error

    # PyTorch reference (float32 throughout)
    ref = torch.nn.functional.layer_norm(x, (N,), w, b, eps)

    y   = layer_norm(x, w, b, eps)

    # If the mutant changes behaviour, this assert will fail.
