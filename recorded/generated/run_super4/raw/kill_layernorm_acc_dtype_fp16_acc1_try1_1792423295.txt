def _test_large_variance_overflow():
    """Variance large enough that float16 accumulation overflows when summed across the block."""
    torch.manual_seed(123)
    M, N = 2, 1024          # N is a multiple of BLOCK_SIZE=1024
    # Create input with large standard deviation (~10) → per‑column sum ≈ 8 * 100 = 800 > 65535 limit of float16
    x = torch.randn(M, N, dtype=torch.float32) * 10.0
    w = torch.rand(N, dtype=torch.float32)
    b = torch.rand(N, dtype=torch.float32)

    # Reference using PyTorch (float32 accumulation)
    ref = torch.nn.functional.layer_norm(x, (N,), w, b, eps=1e-5)

    y = layer_norm(x, w, b, eps=1e-5)

    # Original kernel should match reference within tolerance
