def _check_large_B_tail():
    # Shape with a small remainder in K so many elements are OOB in the last K‑tile
    M, N, K = 64, 64, 65          # K = 2*32 + 1  → 31 OOB elements per row in the last tile
    torch.manual_seed(2024)
    # A = zero → reference result is zero
    a = torch.zeros((M, K), dtype=torch.float16)
    # B = large constant (still finite in float16)
    b = torch.full((K, N), 65500.0, dtype=torch.float16)
    ref = torch.matmul(a, b)       # should be all zeros
    out = matmul(a, b)             # kernel under test
    # Original kernel must match the reference (zero) within tolerance
    assert torch.allclose(out, ref, atol=0.125, rtol=0), \
        f"original kernel failed on shape {M}x{N}x{K}"
    # Mutant kernel adds spurious contributions from the OOB part of A
    # (it loads 1.0 instead of 0.0), producing a non‑zero output.
    assert not torch.allclose(out, ref, atol=0.125, rtol=0), \
        f"mutant not distinguished on shape {M}x{N}x{K}"

_check_large_B_tail()
