def test_large_nonrepresentable_constant():

    With a small N (to avoid overflow) the float16 accumulator incurs a
    noticeable error in the mean, which propagates to the output and exceeds
    the existing tolerance."""

    const = 10000.1                 # not exactly representable in float16 (spacing ~8)
    x = torch.full((M, N), const, dtype=torch.float32)
    w = torch.rand(N, dtype=torch.float32)

    assert torch.allclose(y, expected, atol=1e-2, rtol=0), \
