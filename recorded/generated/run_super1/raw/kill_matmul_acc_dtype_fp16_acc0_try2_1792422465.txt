def _test_large_K_accumulator_error():
    # Use a large inner dimension so that FP16 accumulation error exceeds the test tolerance.
    M, N, K = 64, 64, 1024   # a: (64,1024), b: (1024,64), c: (64,64)
    torch.manual_seed(2025)
    a = torch.randn((M, K), dtype=torch.float16)
    b = torch.randn((K, N), dtype=torch.float16)
    ref = torch.matmul(a, b)          # reference uses FP32 accumulation
    got = matmul(a, b)                # kernel under test (mutant uses FP16 accumulator)
    # Same tolerance as the original sanity check.
    assert torch.allclose(got, ref, atol=0.125, rtol=0), \
