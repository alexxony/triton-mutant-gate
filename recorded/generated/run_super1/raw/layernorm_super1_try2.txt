def test_constant_row_ref(M, N, eps=1e-5):
    """Layer norm of a constant row should match PyTorch's implementation."""
    torch.manual_seed(0)
    x = torch.full((M, N), 3.14, dtype=torch.float32)
    w = torch.rand(N, dtype=torch.float32)
    b = torch.rand(N, dtype=torch.float32)
    y = layer_norm(x, w, b, eps)
    ref = torch.nn.functional.layer_norm(x, (N,), w, b, eps)
    assert torch.allclose(y, ref, atol=1e-5, rtol=0), \
        f"constant row mismatch M={M}, N={N}, eps={eps}"

# Run a few constant‑input tests that exercise boundary/tail handling
