# Additional targeted tests to catch boundary, tail, and stride-related bugs
def check_mn(m, n, scale=1.0, contiguous=True):
    """Run a single softmax correctness test with optional non‑contiguous input."""
    torch.manual_seed(0)
    if contiguous:
        x = torch.randn(m, n, dtype=torch.float32) * scale
    else:
        # Produce a non‑contiguous view by transposing a larger tensor
        base = torch.randn(n, m, dtype=torch.float32) * scale
        x = base.t()  # shape (m, n), strides swapped
    y = softmax(x)
    y_ref = torch.softmax(x, dim=1)
    assert torch.allclose(y, y_ref, atol=1e-5),
        f"softmax mismatch: m={m}, n={n}, scale={scale}, contiguous={contiguous}"
