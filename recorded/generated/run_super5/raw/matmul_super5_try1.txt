# Additional tests targeting boundary/tail handling, non‑contiguous inputs,
# and small/odd shapes that stress the modulo‑based indexing in the kernel.

def _run_one_test(M, N, K, *, seed=0, a=None, b=None):
    """Helper that runs a single correctness check."""
    torch.manual_seed(seed)
    if a is None:
        a = torch.randn((M, K), dtype=torch.float16)
    if b is None:
        b = torch.randn((K, N), dtype=torch.float16)
    ref = torch.matmul(a, b)
    out = matmul(a, b)
    # The kernel uses float32 accumulation then casts to float16;
    # 0.125 absolute tolerance matches the original checker.
    assert torch.allclose(out, ref, atol=0.125, rtol=0),
        f"matmul mismatch for shape ({M},{N},{K})"

# 1. Non‑multiple dimensions in all axes
_run_one_test(70, 70, 70, seed=1)

# 2. M not multiple of BLOCK_SIZE_M, N not multiple of BLOCK_SIZE_N,
#    K not multiple of BLOCK_SIZE_K
_run_one_test(63, 100, 50, seed=2)

# 3. K not multiple of BLOCK_SIZE_K while M,N are multiples
_run_one_test(128, 128, 33, seed=3)

# 4. Minimal problem size (1x1x1)
_run_one_test(1, 1, 1, seed=4)

# 5. Slightly over the block size in each dimension
_run_one_test(65, 65, 65, seed=5)

# 6. Non‑contiguous inputs (transposed views) to stress stride handling
#    Create contiguous base tensors then take a transpose view.
torch.manual_seed(6)
a_base = torch.randn((K, M), dtype=torch.float16)   # will be transposed to (M, K)
b_base = torch.randn((N, K), dtype=torch.float16)   # will be transposed to (K, N)
a = a_base.t()   # shape (M, K), non‑contiguous
b = b_base.t()   # shape (K, N), non‑contiguous
_run_one_test(M, N, K, seed=6, a=a, b=b)
