# Additional targeted tests to catch boundary, tail, and stride-related bugs
def _run_test(m, n, scale=1.0, contiguous=True):
    torch.manual_seed(0)
    if contiguous:
        x = torch.randn(m, n, dtype=torch.float32) * scale
    else:
        # Make a non‑contiguous view by transposing a larger tensor
        base = torch.randn(n, m, dtype=torch.float32) * scale
        x = base.t()          # shape (m, n), strides swapped
    y = softmax(x)
    y_ref = torch.softmax(x, dim=1)
    assert torch.allclose(y, y_ref, atol=1e-5), \
        f"softmax mismatch: m={m}, n={n}, scale={scale}, contiguous={contiguous}"

# ---- Edge‑case matrix sizes ----
# Minimal width
_run_test(2, 1, contiguous=True)
# Power‑of‑two widths
_run_test(2, 2, contiguous=True)
_run_test(2, 1024, contiguous=True)
# Slightly above power‑of‑two (tests tail handling)
_run_test(2, 3, contiguous=True)
_run_test(2, 1025, contiguous=True)
# Non‑contiguous input (stride != element size)
_run_test(2, 1000, contiguous=False)
