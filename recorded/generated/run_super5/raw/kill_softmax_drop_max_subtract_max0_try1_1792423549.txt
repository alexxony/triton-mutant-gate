def test_large_values():
    """Test with large input values that cause overflow if max subtraction is omitted."""
    torch.manual_seed(0)
    # Scale large enough that raw values exceed ~710 (exp overflow in float64)
    x = torch.randn(4, 4, dtype=torch.float32) * 1e4   # typical values ~ few thousand
    y = softmax(x)
    y_ref = torch.softmax(x, dim=1)
    # The original kernel (with max subtraction) should match the reference;
    # the mutant (without subtraction) will produce Inf/NaN and fail this assert.
    assert torch.allclose(y, y_ref, atol=1e-5), (
        f"softmax mismatch on large values: max abs diff = {(y - y_ref).abs().max().item()}"
    )

# Append the new test after the existing sanity checks
